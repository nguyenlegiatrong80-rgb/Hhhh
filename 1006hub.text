-- [[ HỆ THỐNG ÂM THANH & VOICE ]] --
local function Play1006Effect(actionName)
    -- Tiếng cười đặc trưng
    local Laugh = Instance.new("Sound", SoundService)
    Laugh.SoundId = "rbxassetid://6340211440"
    Laugh.Volume = 1.5
    Laugh:Play()
    -- Giọng nói "Đã kích hoạt" (Monster Tone)
    local Voice = Instance.new("Sound", SoundService)
    Voice.SoundId = "rbxassetid://9061036224"
    Voice.Pitch = 0.6
    Voice.Volume = 2
    Voice:Play()
    
    Fluent:Notify({Title = "1006 HUB", Content = "Đã kích hoạt: " .. actionName, Duration = 3})
    game:GetService("Debris"):AddItem(Laugh, 5)
    game:GetService("Debris"):AddItem(Voice, 5)
end

-- LOGIC GOM QUÁI BÀN TAY
local function BringWithHand()
    spawn(function()
        while _G.BringMobs do
            task.wait(1.5)
            pcall(function()
                local Target = game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame * CFrame.new(0,0,-5)
                -- Tạo bàn tay 1006 gom quái
                local Hand = Instance.new("Part", workspace)
                Hand.Size = Vector3.new(15, 2, 15); Hand.Anchored = true; Hand.CanCollide = false
                Hand.Color = Color3.fromRGB(20,20,20); Hand.Material = "Neon"; Hand.CFrame = Target * CFrame.new(0,15,0)
                TweenService:Create(Hand, TweenInfo.new(0.5), {CFrame = Target, Transparency = 1}):Play()
                
                for _, v in pairs(workspace.Enemies:GetChildren()) do
                    if v:FindFirstChild("HumanoidRootPart") and (v.HumanoidRootPart.Position - Target.p).Magnitude <= 1000 then
                        v.HumanoidRootPart.CFrame = Target
                        v.HumanoidRootPart.CanCollide = false
                    end
                end
                game:GetService("Debris"):AddItem(Hand, 0.6)
            end)
        end
    end)
end
